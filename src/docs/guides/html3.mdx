// Function to create a video card
function createVideoCard(video) {
  const tagsHtml = video.tags.map(tag => 
    `<span class="badge ${tag.variant} badge-outline font-semibold">${tag.text}</span>`
  ).join('\n          ');

  return `
    <div data-theme="${video.theme}" class="bg-base-100 shadow-xl border border-base-300 rounded-[14px] w-full max-w-sm overflow-hidden font-sans not-prose card-compact card">
      <a class="block text-inherit no-underline" href="${video.url}" target="_blank">
        <figure class="relative m-0 p-0 h-auto overflow-hidden">
          <img class="block m-0 p-0 w-full max-h-[220px] object-cover" src="${video.thumbnail}" alt="${video.title}" />
        </figure>
        <div class="card-body">
          <div class="flex items-start gap-2.5 mb-2">
            <div class="inline-flex justify-center items-center mt-0.5 text-error">
              <svg class="block w-5 h-5" viewBox="0 0 24 24">
                <rect x="2" y="4" width="20" height="16" rx="4" fill="currentColor"></rect>
                <polygon points="10,8 16,12 10,16" fill="#fff"></polygon>
              </svg>
            </div>
            <div class="flex-1">
              <h3 class="m-0 font-bold text-base text-base-content leading-snug card-title">${video.title}</h3>
              <p class="m-0.5 mt-0 text-xs text-base-content/70">${video.subtitle}</p>
            </div>
          </div>
          <p class="mb-3 text-sm text-base-content/80 leading-relaxed">${video.description}</p>
          <div class="justify-start gap-1.5 mb-1 card-actions">
            ${tagsHtml}
          </div>
        </div>
      </a>
    </div>
  `;
}

// Render cards on page load
document.addEventListener('DOMContentLoaded', function() {
  const container = document.getElementById('video-cards-container');
  if (container) {
    container.innerHTML = videos.map(createVideoCard).join('\n');
  }
});
</script>

## Alternative: Inline data attribute approach

<div id="inline-cards-container" class="flex flex-wrap justify-start gap-6 not-prose">
  <div class="video-card-data" data-video='{"theme":"cupcake","url":"https://www.youtube.com/watch?v=vzlXfZlH5dk","thumbnail":"https://img.youtube.com/vi/vzlXfZlH5dk/maxresdefault.jpg","title":"Main Koi Aisa Geet Gaoon – Yes Boss","subtitle":"Yes Boss · Official Music Video","description":"A classic romantic number featuring SRK & Juhi Chawla — pure 90s Bollywood magic.","tags":[{"text":"Music","variant":"badge-primary"},{"text":"90s Bollywood","variant":"badge-secondary"},{"text":"Romantic","variant":"badge-accent"}]}'></div>
  <div class="video-card-data" data-video='{"theme":"nord","url":"https://www.youtube.com/watch?v=km2OPUctni4","thumbnail":"https://img.youtube.com/vi/km2OPUctni4/maxresdefault.jpg","title":"Dil To Pagal Hai – Title Song","subtitle":"DTPH · 1997 · Official Video","description":"One of the most iconic romantic songs of the 90s.","tags":[{"text":"Classic","variant":"badge-success"},{"text":"Romantic","variant":"badge-error"},{"text":"SRK","variant":"badge-info"}]}'></div>
</div>

<script>
document.addEventListener('DOMContentLoaded', function() {
  const container = document.getElementById('inline-cards-container');
  if (container) {
    const dataElements = container.querySelectorAll('.video-card-data');
    const cardsHtml = Array.from(dataElements).map(el => {
      const video = JSON.parse(el.getAttribute('data-video'));
      return createVideoCard(video);
    }).join('\n');
    container.innerHTML = cardsHtml;
  }
});
</script>
